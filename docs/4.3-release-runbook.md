## 7. zipalign + apksigner (password never printed)

```bash
cd "$REPO/capacitor-app/android/app/build/outputs/apk/release"
PWF="$REPO/android-app/KEYSTORE_PASSWORD_KEEP_SAFE.txt"
KS="$REPO/android-app/android.keystore"
field() { awk -v k="$1" 'index($0,k)==1 { sub(/^[^:]*:[ \t]*/,""); sub(/[ \t\r]+$/,""); print; exit }' "$PWF"; }
export KS_PASS="$(field 'Keystore password:')"
export KEY_PASS="$(field 'Key password:')"
KS_ALIAS="$(field 'Alias:')"
[ -n "$KS_PASS" ] && [ -n "$KEY_PASS" ] && [ -n "$KS_ALIAS" ] && echo "fields ok" || echo "MISSING FIELD - stop"

"$BT/zipalign.exe" -f -p 4 app-release-unsigned.apk aligned.apk
"$BT/apksigner.bat" sign --ks "$KS" --ks-key-alias "$KS_ALIAS" \
  --ks-pass env:KS_PASS --key-pass env:KEY_PASS \
  --out "breaklog-$VER-$CODE.apk" aligned.apk
unset KS_PASS KEY_PASS KS_ALIAS

"$BT/apksigner.bat" verify --print-certs "breaklog-$VER-$CODE.apk" | grep -i 'SHA-256'
# must show: b4fb9ba652fb1302f2fc27c13888aa20a515c3be112f49c2b77d295dc5fa5bbf
"$BT/apksigner.bat" verify --print-certs "breaklog-$VER-$CODE.apk" | grep -qi b4fb9ba652fb1302f2fc27c13888aa20a515c3be112f49c2b77d295dc5fa5bbf && echo "KEY OK" || echo "WRONG KEY - stop"
```

No `set -x`, and do not paste the password file anywhere.
